  ```go
  func (a *Agent) tryReactiveCompactRecovery(...) bool {
      attempt := p.ReactiveCompactAttempt + 1  // 从 payload 读取并升级
      if attempt > a.cfg.MaxReactiveCompactAttempts {
          return false
      }
      // Step 1: call reactive compact callback
      a.cfg.RecoverFromPromptTooLong(ctx, p.SessionID, attempt)
      // Step 2: insert feedback message
      a.cfg.InsertFeedbackMessage(ctx, p.SessionID, turnID, ...)
      // Step 3: FailTurn
      a.cfg.FailTurn(skipCtx, turnID, exitReason)
      // Step 4: publish submit work item (携带 attempt)
      payload := string(MarshalSubmitPayload(p.SessionID, attempt))
      a.queue.Publish(context.Background(), p.SessionID, payload, 100)
      return true
  }
  ```
